cmake_minimum_required(VERSION 3.24)
project(fluid_device_policy_tests LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(fluid_device_policy_tests test_device.c ../device_policy.c ../../core/fl_core.c ../../core/fl_protocol.c)
target_include_directories(fluid_device_policy_tests PRIVATE .. ../../core)
add_executable(fluid_usb_budget_tests test_usb_budget.c ../usb_task.c)
target_include_directories(fluid_usb_budget_tests PRIVATE usb_stub)
target_compile_definitions(fluid_usb_budget_tests PRIVATE FL_TINYUSB_USBD_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}/usb_stub/usbd.c")
add_executable(fluid_hal_tests test_hal.c ../fl_hal.c)
target_include_directories(fluid_hal_tests PRIVATE hal_stub .. ../../core)
if(MSVC)
  target_compile_options(fluid_device_policy_tests PRIVATE /W4 /WX)
  target_compile_options(fluid_usb_budget_tests PRIVATE /W4 /WX)
  target_compile_options(fluid_hal_tests PRIVATE /W4 /WX)
else()
  target_compile_options(fluid_device_policy_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(fluid_usb_budget_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(fluid_hal_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(fluid_device_policy_tests PRIVATE m)
endif()
enable_testing()
add_test(NAME device_policy COMMAND fluid_device_policy_tests)
add_test(NAME usb_task_budget COMMAND fluid_usb_budget_tests)
add_test(NAME adc_quality_and_bounds COMMAND fluid_hal_tests)
