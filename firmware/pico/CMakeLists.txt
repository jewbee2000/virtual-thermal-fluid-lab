cmake_minimum_required(VERSION 3.24)
if(NOT PICO_SDK_PATH)
  message(FATAL_ERROR "Pass the pinned Pico SDK2.3.1 path; no automatic dependency fetch")
endif()
set(PICO_BOARD pico CACHE STRING "RP2040 Pico target")
if(NOT PICO_BOARD STREQUAL "pico")
  message(FATAL_ERROR "This indicator/pin contract targets Pico, not Pico W or another board")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(fluid_controller_pico VERSION 0.1.0 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
pico_sdk_init()
if(NOT PICO_SDK_VERSION_STRING VERSION_EQUAL "2.3.1")
  message(FATAL_ERROR "Pico SDK must be2.3.1")
endif()
set(fl_usbd_source "${PICO_SDK_PATH}/lib/tinyusb/src/device/usbd.c")
file(SHA256 "${fl_usbd_source}" fl_usbd_sha256)
if(NOT fl_usbd_sha256 STREQUAL "d198d57e626db62c2ac42fe9c26c815403984d92cedecd5414a9b38d3836a95d")
  message(FATAL_ERROR "TinyUSB usbd.c hash changed; review the bounded task wrapper")
endif()
file(READ "${fl_usbd_source}" fl_usbd_text)
string(REGEX MATCHALL "osal_queue_receive\\(_usbd_q, &event, timeout_ms\\)" fl_receive_sites "${fl_usbd_text}")
list(LENGTH fl_receive_sites fl_receive_site_count)
if(NOT fl_receive_site_count EQUAL 1)
  message(FATAL_ERROR "Expected one reviewed TinyUSB receive site")
endif()
get_target_property(fl_usb_sources tinyusb_device_base INTERFACE_SOURCES)
list(REMOVE_ITEM fl_usb_sources "${fl_usbd_source}")
set_property(TARGET tinyusb_device_base PROPERTY INTERFACE_SOURCES "${fl_usb_sources}")
set_source_files_properties(main.c fl_hal.c device_policy.c usb_descriptors.c usb_task.c
  "${CMAKE_CURRENT_SOURCE_DIR}/../core/fl_core.c" "${CMAKE_CURRENT_SOURCE_DIR}/../core/fl_protocol.c"
  PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Wpedantic;-Werror")
function(fl_device_target target mode)
  add_executable(${target} main.c fl_hal.c device_policy.c usb_descriptors.c usb_task.c ../core/fl_core.c ../core/fl_protocol.c)
  target_include_directories(${target} PRIVATE . ../core)
  target_compile_definitions(${target} PRIVATE FL_DEVICE_MODE=${mode})
  target_compile_definitions(${target} PRIVATE FL_TINYUSB_USBD_SOURCE="${fl_usbd_source}")
  target_link_libraries(${target} PRIVATE pico_stdlib hardware_adc hardware_watchdog pico_unique_id tinyusb_device m)
  pico_enable_stdio_usb(${target} 0)
  pico_enable_stdio_uart(${target} 0)
  target_link_options(${target} PRIVATE -Wl,--print-memory-usage)
  pico_set_program_name(${target} "Virtual Thermal Fluid Lab")
  pico_set_program_version(${target} "0.1.0")
  pico_add_extra_outputs(${target})
endfunction()
fl_device_target(fluid_controller_pico_link 1)
fl_device_target(fluid_controller_pico_peripheral 2)
