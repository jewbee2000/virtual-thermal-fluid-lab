cmake_minimum_required(VERSION 3.24)
project(virtual_thermal_fluid_lab VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(FL_SANITIZE "Enable GCC/Clang address and undefined behavior checks" OFF)
add_library(fl_core STATIC firmware/core/fl_core.c firmware/core/fl_protocol.c)
target_include_directories(fl_core PUBLIC firmware/core)
if(MSVC)
  target_compile_options(fl_core PUBLIC /W4 /WX)
else()
  target_compile_definitions(fl_core PUBLIC _POSIX_C_SOURCE=200809L)
  target_compile_options(fl_core PUBLIC -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(fl_core PUBLIC m)
  if(FL_SANITIZE)
    target_compile_options(fl_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(fl_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(fluid_controller_host firmware/host/main.c)
target_link_libraries(fluid_controller_host PRIVATE fl_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(fl_core_tests firmware/tests/test_core.c)
  target_link_libraries(fl_core_tests PRIVATE fl_core)
  add_test(NAME portable_core_codec COMMAND fl_core_tests)
endif()
